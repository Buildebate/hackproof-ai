import {
  createApp,
  createSSRApp,
  defineComponent,
  h,
  nextTick,
  ref,
  useCssVars,
} from 'vue'
import { renderToString } from '@vue/server-renderer'

const Child = defineComponent({
  props: { rootStyle: { type: null, default: undefined } },
  setup(props: any, { slots }: any) {
    return () =>
      h(
        'div',
        { class: 'root', style: props.rootStyle },
        slots.default ? slots.default() : [],
      )
  },
})

function parentWith(
  vars: () => Record<string, unknown>,
  rootStyle: any,
  slot?: string,
) {
  return defineComponent({
    setup() {
      useCssVars(vars as any)
      return () =>
        h(Child, { rootStyle }, slot === undefined ? undefined : () => slot)
    },
  })
}

function mount(comp: any, props?: any) {
  const container = document.createElement('div')
  createApp(comp, props).mount(container)
  return container.querySelector('.root') as HTMLElement
}

describe('cssVars fallthrough to child root', () => {
  test('default blue for undefined, removal for null/empty', async () => {
    const WithDefault = defineComponent({
      props: {
        backgroundColor: { type: String, default: 'blue' },
        padding: { type: String, default: undefined },
      },
      setup(props: any) {
        useCssVars(
          () =>
            ({
              'case-color': props.backgroundColor,
              'case-padding': props.padding,
            }) as any,
        )
        return () => h(Child, { rootStyle: { display: 'block' } })
      },
    })
    expect(
      mount(WithDefault, {}).style.getPropertyValue('--case-color'),
    ).toBe('blue')
    expect(
      mount(WithDefault, { backgroundColor: undefined }).style.getPropertyValue(
        '--case-color',

    ).toBe('blue')
    const nullEl = mount(WithDefault, { backgroundColor: null })
    expect(nullEl.style.getPropertyValue('--case-color')).toBe('')
    expect(nullEl.style.cssText).not.toContain('--case-padding:')

    const color = ref<any>('red')
    const padding = ref<any>('')
    const el = mount(
      parentWith(
        () => ({ 'case-color': color.value, 'case-padding': padding.value }),
        { display: 'block' },
      ),
    )
    expect(el.style.getPropertyValue('--case-padding')).toBe('')
    padding.value = 0
    await nextTick()
    expect(el.style.getPropertyValue('--case-padding')).toBe('0')
    padding.value = 20
    await nextTick()
    expect(el.style.getPropertyValue('--case-padding')).toBe('20')

    expect(el.style.getPropertyValue('--case-color')).toBe('123')
  })

  test('child :style null/{} /[]/"" preserves vars; display true/0 dropped', async () => {

    for (const next of [null, {}, [], ''] as any[]) {
