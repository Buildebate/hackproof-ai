import { describe, it, expect } from 'vitest'
import { compileStyle } from '@vue/compiler-sfc'
describe('scoped nested at-rule regression', () => {
  it('wraps declarations inside nested @media with scoped nesting selector', async () => {
    const source = `.foo {\n  background-color: red;\n  .bar {}\n  @media only screen and (max-width: 800px) {\n    background-color: green;\n  }\n}`;
    const { code } = await compileStyle({
      source,
      filename: 'test.vue',
      id: 'data-v-7ba5bd90',
      scoped: true,
    });
    const matches = code.match(/&\[data-v-7ba5bd90\]/g) || [];
    expect(matches.length).toBe(2);
  });
});
